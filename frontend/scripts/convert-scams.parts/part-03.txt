  "social_dm",
        "dating_app",
        "marketplace",
        "job_board",
        "web",
      ].includes(channel)
    ) {
      set.add("web");
    } else {
      set.add("other");
    }
  }
  return set.size ? [...set] : ["other"];
}

function demandsFromAsks(asks) {
  const set = new Set();
  const blob = (asks ?? []).join(" ").toLowerCase();
  if (blob.includes("gift")) set.add("gift_card");
  if (/wire|bank_transfer/.test(blob)) set.add("wire");
  if (blob.includes("crypto")) set.add("crypto");
  if (blob.includes("check")) set.add("check");
  if (/zelle|venmo|cash|payment_app/.test(blob)) set.add("cash");
  if (set.size === 0) set.add("other");
  return [...set];
}

function victimRolesFromCatalog(entry) {
  const roles = [];
  const blob = `${entry.id} ${entry.name} ${(entry.tags?.hooks ?? []).join(" ")}`.toLowerCase();
  if (blob.includes("romance") || blob.includes("dating")) roles.push("dating app user");
  if (blob.includes("job") || blob.includes("recruiter") || blob.includes("mule")) roles.push("job seeker");
  if (blob.includes("marketplace") || blob.includes("rental") || blob.includes("pet")) {
    roles.push("marketplace buyer", "marketplace seller");
  }
  if (blob.includes("medicare") || blob.includes("ssa") || blob.includes("burial")) {
    roles.push("older adult");
  }
  if (blob.includes("irs") || blob.includes("tax") || blob.includes("government")) roles.push("taxpayer");
  if (blob.includes("bank")) roles.push("bank customer");
  if (roles.length === 0) roles.push("consumer");
  return unique(roles);
}

function howItWorksSteps(how) {
  return String(how ?? "")
    .split(/\s*\d+\)\s*/)
    .map((part) => part.trim().replace(/\.\s*$/, "."))
    .filter(Boolean);
}
