function adviceFor(scam) {
  const blob = `${scam.slug} ${scam.description} ${scam.demands.join(" ")} ${(scam.catalogAdvice ?? []).join(" ")}`.toLowerCase();
  const steps = [...(scam.catalogAdvice ?? [])];
  const push = (line) => {
    if (!steps.some((step) => step.toLowerCase() === line.toLowerCase())) steps.push(line);
  };

  push("Stop the conversation. Do not send more money, codes, or remote-access access.");
  push("Verify through a number or website you look up yourself — never one they provided.");
  push("Report it at reportfraud.ftc.gov. If money moved, call your bank or card issuer now.");

  if (scam.demands.includes("gift_card")) {
    push("Gift-card loads are treated like cash. Tell the store and the card issuer immediately; reversal is rare.");
  }
  if (scam.demands.includes("crypto")) {
    push("Crypto transfers are usually irreversible. Save wallet addresses, TXIDs, and chat logs for a report.");
  }
  if (scam.demands.includes("wire")) {
    push("Call the sending bank’s fraud line and ask them to recall the wire. Speed matters more than courtesy.");
  }
  if (scam.demands.includes("check")) {
    push("A deposited check can bounce days later. Do not spend or “refund” any of it. Ask the bank to put a hold and file fraud.");
  }
  if (blob.includes("remote") || blob.includes("anydesk") || blob.includes("teamviewer")) {
    push("Disconnect from the internet, uninstall the remote tool, and change passwords from a different device.");
  }
  if (blob.includes("otp") || blob.includes("seed phrase") || blob.includes("password")) {
    push("Treat shared codes or seed phrases as a full account compromise. Reset credentials and enable a new 2FA method.");
  }
  if (blob.includes("romance") || blob.includes("dating") || blob.includes("sugar") || blob.includes("e_dating")) {
    push("Tell someone you trust. E-dating and “investment mentor” cons rely on isolation and urgency.");
  }
  if (blob.includes("job") || blob.includes("training") || blob.includes("mule")) {
    push("Real employers do not charge you to get hired, and they do not pay you to cash checks or reship parcels.");
  }
  if ((scam.reportTo ?? []).length) {
    push(`Agencies that take this: ${scam.reportTo.join(", ")}.`);
  }

  return steps.slice(0, 8);
}
