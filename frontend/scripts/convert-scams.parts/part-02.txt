function titleFromSlug(slug) {
  if (TITLE_OVERRIDES[slug]) return TITLE_OVERRIDES[slug];
  return slug
    .split("_")
    .map((part) => {
      if (ACRONYMS.has(part)) return part.toUpperCase();
      return part.charAt(0).toUpperCase() + part.slice(1);
    })
    .join(" ");
}

function unescapeTsvCell(raw) {
  let text = (raw ?? "").trim();
  if (text.startsWith('"') && text.endsWith('"')) {
    text = text.slice(1, -1).replace(/""/g, '"');
  }
  return text;
}

function parsePgArray(raw) {
  let text = unescapeTsvCell(raw);
  if (!text || text === "{}") return [];
  if (text.startsWith("{") && text.endsWith("}")) {
    text = text.slice(1, -1);
  }
  if (!text.trim()) return [];
  const items = [];
  const re = /"((?:[^"\\]|\\.)*)"|([^,{}]+)/g;
  let match;
  while ((match = re.exec(text))) {
    const value = (match[1] ?? match[2] ?? "").replace(/\\"/g, '"').trim();
    if (value && value !== "{" && value !== "}") items.push(value);
  }
  return items;
}

function parseTsv(text) {
  const lines = text.replace(/^\uFEFF/, "").split(/\r?\n/).filter(Boolean);
  const headers = lines[0].split("\t");
  return lines.slice(1).map((line) => {
    const cols = line.split("\t");
    const row = {};
    headers.forEach((header, i) => {
      row[header] = cols[i] ?? "";
    });
    return row;
  });
}

function emptyTags() {
  return { channels: [], hooks: [], asks: [], signals: [] };
}

function mergeTags(a, b) {
  const out = emptyTags();
  for (const group of Object.keys(out)) {
    out[group] = [...new Set([...(a?.[group] ?? []), ...(b?.[group] ?? [])])];
  }
  return out;
}

function unique(list) {
  return [...new Set((list ?? []).filter(Boolean))];
}

function catalogUuid(_slug, index) {

  return `7c2a9f10-4b3e-4d8a-9f21-e0d4c000${n}`.slice(0, 36);
}
